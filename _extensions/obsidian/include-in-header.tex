% ---- Palette ----
% Monochrome brand identity: near-black primary, drawn from the
% website brand colour (#121212). The greys are the functional
% monochrome scale proven by GOV.UK and NHS design systems (text,
% secondary, hairline); the primary is our own brand value.
\definecolor{obsidian}{RGB}{18,18,18}    % primary text (16.8:1 on white)
\definecolor{obsidianl}{RGB}{72,73,73}   % secondary text (9.0:1, passes AAA)
\definecolor{obsidianlink}{RGB}{72,73,73} % links, shown by underline (9.0:1)
\definecolor{ruled}{RGB}{206,206,206}   % hairlines and rules (decorative)
\definecolor{cream}{RGB}{243,243,243}   % off-white surface

% ---- Link colours ----
% Quarto emits its own \hypersetup after this header, so a bare
% \hypersetup here would be overridden. \AtBeginDocument runs at
% \begin{document}, after every preamble call, so these values win.
\AtBeginDocument{%
  \hypersetup{colorlinks=true, linkcolor=obsidianlink, urlcolor=obsidianlink, citecolor=obsidianlink}%
}

% ---- Monochrome callout borders ----
% Quarto hardcodes the per-type callout colours in its Lua filter and
% ignores callout-color for PDF. Override the frame colours at
% \begin{document}, after Quarto's \definecolor calls, so the warning
% callout border is near-black instead of amber.
\AtBeginDocument{%
  \colorlet{quarto-callout-warning-color-frame}{obsidian}%
  \colorlet{quarto-callout-color-frame}{obsidian}%
}

% ---- Contents heading ----
% Quarto renders the table of contents heading outside titlesec, so
% it falls back to black. Fix it to the brand primary, and match the
% list of tables and figures headings.
\AtBeginDocument{%
  \renewcommand{\contentsname}{\textcolor{obsidian}{Contents}}%
  \renewcommand{\listfigurename}{\textcolor{obsidian}{List of Figures}}%
  \renewcommand{\listtablename}{\textcolor{obsidian}{List of Tables}}%
}

% ---- Pagination floor: no widows, orphans, or dangling headings ----
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
\raggedbottom
% Keep section headings with the first lines of their body.
\pretocmd{\section}{\needspace{5\baselineskip}}{}{}
\pretocmd{\subsection}{\needspace{3\baselineskip}}{}{}

% ---- Pin tables in place so a heading and its table cannot split ----
\floatplacement{table}{H}
\BeforeBeginEnvironment{table}{\nopagebreak}

% ---- Section headings: letterspaced, thin rule under sections ----
\titleformat{\section}
  {\Large\bfseries\sffamily\color{obsidian}\lsstyle}
  {\thesection.}{0.6em}{}
  [\vspace{0.15em}{\color{obsidian}\titlerule[0.8pt]}]
\titleformat{\subsection}
  {\large\bfseries\sffamily\color{obsidianl}\lsstyle}
  {\thesubsection.}{0.6em}{}
\titleformat{\subsubsection}
  {\normalsize\bfseries\sffamily\color{obsidianl}\lsstyle}
  {\thesubsubsection.}{0.6em}{}
\titlespacing*{\section}{0pt}{2.2ex}{1.5ex}
\titlespacing*{\subsection}{0pt}{1.6ex}{0.9ex}
\titlespacing*{\subsubsection}{0pt}{1.4ex}{0.8ex}

% Keep the current section name in the running header.
\renewcommand{\sectionmark}[1]{\markright{\thesection.~{\sffamily#1}}}

% ---- Captions: small, bold slate label ----
\captionsetup{font={small}, labelfont={bf,color=obsidianl}, labelsep=colon}

% The TOC must sit on its own page. tocloft redefines \tableofcontents
% in its own \AtBeginDocument (after this preamble), so this wrapper
% must register after tocloft's or it would be clobbered.
% Front matter (abstract, contents) is roman-numbered. The arabic
% switch lives in partials/toc.tex, after the lists, so list of
% tables and figures also stay roman (JSP 101 layout).
\AtBeginDocument{%
  \let\oldtoc\tableofcontents
  \renewcommand{\tableofcontents}{\clearpage\thispagestyle{fancy}\oldtoc\clearpage}%
}
