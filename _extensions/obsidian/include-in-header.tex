% TeX Live 2026 xcolor-patches-tmp-ltx.sty can leak expl3 colour
% commands (e.g. \__color_backend_raw_rgb:n) literally into PDF
% content streams.  Patch the stack push AND the raw RGB setter so
% hyperref link colours (used in TOC dot leaders, citations, and
% cross-references) do not leak raw floats like "0.28235 0.28627".
\begingroup\ExplSyntaxOn
\cs_gset_protected:Npn \__kernel_color_backend_stack_push:nn #1#2
  {
    \tex_pdfextension:D colorstack ~
      \int_eval:n { #1 } ~ push ~ { \expanded{#2} }
  }
% hyperref writes link colours through this path; force \expanded
% so the RGB triplet becomes a valid PDF operator sequence.
\cs_gset_protected:Npn \__color_backend_raw_rgb:n #1
  {
    \tex_pdfextension:D colorstack ~
      0 ~ push ~ { \expanded{ #1 } }
  }
\ExplSyntaxOff
\endgroup

% ---- Link colours ----
% Quarto emits its own \hypersetup after this header, so a bare
% \hypersetup here would be overridden. \AtBeginDocument runs at
% \begin{document}, after every preamble call, so these values win.
% Use black for linkcolor to avoid the TeX Live 2026 l3backend color
% leak (raw RGB floats like 0.28235 appearing as visible text in TOC).
% Body headings use \color{obsidianl} via \titleformat, not linkcolor.
\AtBeginDocument{%
  \hypersetup{colorlinks=true, linkcolor=black, urlcolor=black, citecolor=black}%
}

% ---- Monochrome callout borders ----
% Quarto hardcodes the per-type callout colours in its Lua filter and
% ignores callout-color for PDF. Override the frame colours at
% \begin{document}, after Quarto's \definecolor calls, so the warning
% callout border is near-black instead of amber.
\AtBeginDocument{%
  \colorlet{quarto-callout-warning-color-frame}{obsidian}%
  \colorlet{quarto-callout-color-frame}{obsidian}%
}

% ---- Section headings: letterspaced, thin rule under sections ----
% Numbering: Quarto's number-sections: true emits "N. Title" in the

  {\large\bfseries\obsidianheadfont\color{obsidianl}\lsstyle}
  {}{0em}{}

  {}{0em}{}

\titlespacing*{\subsection}{0pt}{1.6ex}{0.9ex}
\titlespacing*{\subsubsection}{0pt}{1.4ex}{0.8ex}

% The TOC and similar pages use the plain page style by default.
% Make it match so every page carries the same marking. Format
% documents (letter, memo, invoice, agenda) use a minimal variant:
% the classification band stays (the marking contract) but the
% report identity line and page number drop out, because the
% document carries its own page-1 identity (letterhead, masthead
% or header block). \docformatplain is set in before-body.tex.
\fancypagestyle{plain}{%
  \fancyhf{}%
  \ifdefstring{\docformatplain}{1}{%
    \fancyhead[L]{\parbox[b][1.05cm][c]{\textwidth}{%
      {\centering\classification\par}}}%
    \fancyfoot[C]{\parbox[b][0.9cm][c]{\textwidth}{%
      {\centering\classification\par}}}%
  }{%
    \fancyhead[L]{\parbox[b][1.05cm][c]{\textwidth}{%

       \small\color{obsidianl}\docref{} ~\textbar{}~ \doctitle}%
      \hfill{\small\color{obsidianl}\thepage\ / \ztotpages}%
    }}%
    \fancyhead[C]{}
    \fancyhead[R]{}
    \fancyfoot[C]{\parbox[b][0.9cm][c]{\textwidth}{%
      {\small\color{obsidianl}Obsidian Solutions ~\textbar{}~ obsidiansolutions.co.uk%

% The TOC must sit on its own page. tocloft redefines \tableofcontents
% in its own \AtBeginDocument (after this preamble), so this wrapper
% must register after tocloft's or it would be clobbered.
% Front matter (abstract, contents) is roman-numbered. The arabic
% switch lives in partials/toc.tex, after the lists, so list of
% tables and figures also stay roman (JSP 101 layout).
\AtBeginDocument{%
  \let\oldtoc\tableofcontents
  \renewcommand{\tableofcontents}{\clearpage\thispagestyle{fancy}\oldtoc\clearpage}%
}
